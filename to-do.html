// dsPIC33EP Controller for Automated Vacuum System // Target Compiler: XC16 #define FCY 3685000UL // System instruction clock for delay functions #include #include #include #include // --- Pin Mappings & Macros --- #define SENSOR_DIRT PORTAbits.RA0 #define SENSOR_OBSTACLE PORTAbits.RA1 #define BTN_START PORTBbits.RB0 #define LED_RED LATBbits.LATB15 #define LED_GREEN LATBbits.LATB14 // --- System Modes (FSM States) --- typedef enum { MODE_HALTED = 0, // Standby: Both LEDs off MODE_NORMAL, // Cleaning: Green LED on MODE_EVADE, // Obstacle: Red LED on MODE_HEAVY // Dirty Patch: Both LEDs on } SysMode_t; volatile SysMode_t activeMode = MODE_HALTED; // --- Hardware Reset Interrupt (INT1) --- void __attribute__((__interrupt__, no_auto_psv)) _INT1Interrupt(void) { // 1. Force system back to standby state activeMode = MODE_HALTED; // 2. Turn off indicators immediately LED_RED = 0; LED_GREEN = 0; // 3. Clear interrupt flag to prevent re-triggering IFS1bits.INT1IF = 0; } // --- System Setup Routine --- void initHardware(void) { // Disable analog functions on PORTA and PORTB ANSELA = 0x0000; ANSELB = 0x0000; // Configure Input Pins (1 = Input) TRISAbits.TRISA0 = 1; // Dirt sensor TRISAbits.TRISA1 = 1; // Obstacle sensor TRISBbits.TRISB0 = 1; // Start button // Configure Output Pins (0 = Output) TRISBbits.TRISB15 = 0; // Red LED TRISBbits.TRISB14 = 0; // Green LED // Default state is OFF LED_RED = 0; LED_GREEN = 0; // External Interrupt Setup (INT1 on Peripheral Pin 33) RPINR0bits.INT1R = 33; INTCON2bits.INT1EP = 1; // Trigger on falling edge IFS1bits.INT1IF = 0; // Clear initial flag IEC1bits.INT1IE = 1; // Enable INT1 interrupt } int main(void) { // Initialize registers and I/O initHardware(); // --- Main FSM Loop --- while(1) { // 1. Read active-low inputs (A low signal means the sensor/button is active) bool start_pressed = (BTN_START == 0); bool obstacle_detected = (SENSOR_OBSTACLE == 0); bool dirt_detected = (SENSOR_DIRT == 0); // 2. Evaluate State Transitions switch(activeMode) { case MODE_HALTED: if (start_pressed) { activeMode = MODE_NORMAL; } break; case MODE_NORMAL: if (start_pressed) { activeMode = MODE_HALTED; } else if (obstacle_detected) { activeMode = MODE_EVADE; } else if (dirt_detected) { activeMode = MODE_HEAVY; } break; case MODE_EVADE: if (start_pressed) { activeMode = MODE_HALTED; } else if (obstacle_detected) { activeMode = MODE_EVADE; // Remain in evade } else { activeMode = MODE_NORMAL; // Obstacle cleared } break; case MODE_HEAVY: if (start_pressed) { activeMode = MODE_HALTED; } else if (!dirt_detected) { activeMode = MODE_NORMAL; // Dirt cleared } // Else remain in heavy cleaning mode break; default: activeMode = MODE_HALTED; // Fallback break; } // 3. Update Hardware Outputs if (activeMode == MODE_HALTED) { LED_RED = 0; LED_GREEN = 0; } else if (activeMode == MODE_NORMAL) { LED_RED = 0; LED_GREEN = 1; } else if (activeMode == MODE_EVADE) { LED_RED = 1; LED_GREEN = 0; } else if (activeMode == MODE_HEAVY) { LED_RED = 1; LED_GREEN = 1; } // 4. Execution Delay __delay_ms(1000); } return 0; }
