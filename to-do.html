#define FCY 3685000UL #include "xc.h" #include #include "libpic30.h" #pragma config ICS = PGD2 typedef enum { S0, S1, S2, S3 } StateType; volatile StateType current_state = S0; int main(void) { // 1. Setup Pins ANSELAbits.ANSA0 = 0; ANSELAbits.ANSA1 = 0; ANSELBbits.ANSB0 = 0; ANSELBbits.ANSB1 = 0; TRISAbits.TRISA0 = 1; TRISAbits.TRISA1 = 1; TRISBbits.TRISB0 = 1; TRISBbits.TRISB1 = 1; TRISBbits.TRISB15 = 0; TRISBbits.TRISB14 = 0; // 2. Enable Internal Pull-ups CNPUAbits.CNPUA0 = 1; CNPUAbits.CNPUA1 = 1; CNPUBbits.CNPUB0 = 1; CNPUBbits.CNPUB1 = 1; LATBbits.LATB15 = 0; LATBbits.LATB14 = 0; while(1) { // Read Inputs uint16_t input = (!PORTBbits.RB0 << 2) | (!PORTAbits.RA1 << 1) | (!PORTAbits.RA0); // State Machine switch(current_state) { case S0: LATBbits.LATB15 = 0; LATBbits.LATB14 = 0; switch(input) { case 4: case 5: case 6: case 7: current_state = S1; break; default: current_state = S0; break; } break; case S1: LATBbits.LATB15 = 0; LATBbits.LATB14 = 1; switch(input) { case 4: case 5: case 6: case 7: current_state = S0; break; case 2: case 3: current_state = S2; break; case 1: current_state = S3; break; case 0: current_state = S1; break; } break; case S2: LATBbits.LATB15 = 1; LATBbits.LATB14 = 0; switch(input) { case 4: case 5: case 6: case 7: current_state = S0; break; case 2: case 3: current_state = S2; break; case 0: case 1: current_state = S1; break; } break; case S3: LATBbits.LATB15 = 1; LATBbits.LATB14 = 1; switch(input) { case 4: case 5: case 6: case 7: current_state = S0; break; case 1: case 3: current_state = S3; break; case 0: case 2: current_state = S1; break; } break; } // Tiny delay so the chip doesn't flicker wildly __delay_ms(50); ClrWdt(); } return 0; }
